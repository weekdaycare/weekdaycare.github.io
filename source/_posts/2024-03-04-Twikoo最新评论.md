## 新建 twikoo_new.js 脚本

> 基于 1.27.0 ，写的一坨，佬轻喷😭

新建 `twikoo_new.js` 位于 `theme/stellar/source/js/services/twikoo_new.js` ，写入以下内容

```js twikoo_new.js
utils.jq(() => {
    // 加载 Twikoo 的 JavaScript 脚本
    const twikooScript = document.createElement('script');
    twikooScript.src = 'https://npm.onmicrosoft.cn/twikoo@1.6.31';
    twikooScript.async = true;
    document.head.appendChild(twikooScript);

    twikooScript.onload = function() {
        // 当 Twikoo 脚本加载完成后执行加载评论的函数
        $(function () {
            const twikoos = document.getElementsByClassName('ds-twikoo');
            for (var i = 0; i < twikoos.length; i++) {
                const el = twikoos[i];
                const api = el.getAttribute('api');
                const limit = parseInt(el.getAttribute('limit')) || 10;
                const reply = el.getAttribute('hide') === 'reply' ? false : true;
                if (api == null) {
                    continue;
                }
                twikoo.getRecentComments({
                    envId: api,
                    pageSize: limit,
                    includeReply: reply
                }).then(function (res) {
                    for (var j = 0; j < res.length; j++) {
                        var cell = '<div class="timenode" index="' + j + '">';
                        cell += '<div class="header">';
                        cell += '<div class="user-info">';
                        cell += '<span>' + res[j].nick + '</span>';
                        cell += '</div>';
                        cell += '<span>' + res[j].relativeTime + '</span>';
                        cell += '</div>';
                        cell += '<a class="body" href="' + res[j].url + '">';
                        cell += res[j].commentText;
                        cell += '</a>';
                        cell += '</div>';
                        $(el).append(cell);
                    }
                }).catch(function (err) {
                    // 发生错误
                    console.error(err);
                });
            }
        });
    };
});
```

## 主题配置中引入

Stellar {% mark 1.27.0 %} 版本，内置服务配置转移到 `data_services` 中，在主题配置文件的 `data_services` 下引入脚本

```yml _config.stellar.yml
data_services:
  ...
  twikoo:
    js: /js/services/twikoo_new.js
```

至此配置完成

## 如何使用

在 `_data/widgets.yml` 中创建小组件，示例

```yml
new_comment:
  layout: timeline
  title: 最新评论
  api: https://comment.weekdaycare.cn # 此处为 Twikoo 函数
  type: twikoo
  limit: 16 # 限制获取数量，默认为 10
  hide: reply # 是否隐藏回复，不填默认不隐藏
```

然后就可以在主题配置文件中 `site_tree` 里引入小组件了。

## 历史上的今天

同样是使用 timeline 组件展示的历史上的今天

```js history.js
utils.jq(() => {
	$(function () {
	  const els = document.getElementsByClassName('ds-history');
	  for (var i = 0; i < els.length; i++) {
		const el = els[i];
		const api = el.getAttribute('api');
		if (api == null) {
		  continue;
		}
		// layout
		utils.request(el, api, function(data) {
		  const list = data.data.list || [];
		  const limit = el.getAttribute('limit') || list.length;
		  for (let i = 0; i < limit; i++) {
			const item = list[i];
			var cell = '<div class="timenode" index="' + i + '">';
			cell += '<div class="header">';
			cell += '<div class="user-info">';
			cell += '<span>' + '历史上' + '</span>';
			cell += '</div>';
			cell += '<span>' + item.year + '年' + '</span>';
			cell += '</div>';
			cell += '<a class="body" href="' + item.url + '" target="_blank" rel="external nofollow noopener noreferrer">';
			cell += item.title;
			cell += '</a>';
			cell += '</div>';
			$(el).append(cell);
		  }
		});
	  }
	});
  });
```

使用方法大致和上面的相同，调用的 `api` 是 https://tenapi.cn/v2/history

```yaml widgets.yml
history:
  layout: timeline
  title: 历史上的今天
  api: https://tenapi.cn/v2/history
  type: history
  limit: 3
```
